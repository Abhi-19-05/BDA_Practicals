
-- 1. Hive Configuration
hive
SET hive.support.concurrency = true;
SET hive.enforce.bucketing = true;
SET hive.exec.dynamic.partition.mode = nonstrict;
SET hive.txn.manager = org.apache.hadoop.hive.ql.lockmgr.DbTxnManager;
SET hive.compactor.initiator.on = true;


-- 2. Create Transactions Table

CREATE TABLE transactions (
    txn_id INT,
    account_id INT,
    txn_type STRING,
    amount FLOAT,
    branch STRING,
    txn_date STRING
)
PARTITIONED BY (txn_year INT)
CLUSTERED BY (account_id) INTO 4 BUCKETS
STORED AS ORC
TBLPROPERTIES ("transactional"="true");


-- 3. Create Accounts Table

CREATE TABLE accounts (
    account_id INT,
    customer_name STRING,
    account_type STRING,
    branch STRING,
    opening_date STRING
)
CLUSTERED BY (account_id) INTO 4 BUCKETS
STORED AS ORC
TBLPROPERTIES ("transactional"="true");

-- 12. Inner Join

SELECT
    t.txn_id,
    t.account_id,
    a.customer_name,
    a.account_type,
    t.txn_type,
    t.amount,
    t.branch,
    t.txn_date,
    t.txn_year
FROM transactions t
JOIN accounts a
ON t.account_id = a.account_id;


-- 13. Left Join: Accounts Without 2025 Transactions

SELECT
    a.account_id,
    a.customer_name,
    a.account_type,
    a.branch
FROM accounts a
LEFT JOIN transactions t
ON a.account_id = t.account_id
AND t.txn_year = 2025
WHERE t.account_id IS NULL;


-- 14. Total Amount by Branch, Year and Transaction Type

SELECT
    branch,
    txn_year,
    txn_type,
    SUM(amount) AS total_amount
FROM transactions
GROUP BY branch, txn_year, txn_type
ORDER BY txn_year, branch, txn_type;


-- 15. Top 5 Accounts by Total Transaction Amount

SELECT
    a.account_id,
    a.customer_name,
    SUM(t.amount) AS total_transaction_amount
FROM transactions t
JOIN accounts a
ON t.account_id = a.account_id
GROUP BY a.account_id, a.customer_name
ORDER BY total_transaction_amount DESC
LIMIT 5;


-- 16. Suspicious Withdrawal Activity

SELECT
    account_id,
    txn_date,
    COUNT(*) AS withdrawal_count
FROM transactions
WHERE txn_type = 'Withdrawal'
GROUP BY account_id, txn_date
HAVING COUNT(*) > 10
ORDER BY withdrawal_count DESC;


-- 17. Year-over-Year Transaction Growth Using LAG

SELECT
    branch,
    txn_year,
    total_amount,
    LAG(total_amount, 1) OVER (
        PARTITION BY branch
        ORDER BY txn_year
    ) AS previous_year_amount,
    total_amount - LAG(total_amount, 1) OVER (
        PARTITION BY branch
        ORDER BY txn_year
    ) AS yoy_growth
FROM (
    SELECT
        branch,
        txn_year,
        SUM(amount) AS total_amount
    FROM transactions
    GROUP BY branch, txn_year
) yearly_data
ORDER BY branch, txn_year;


-- 18. Branch with Highest Average Transaction Amount

SELECT
    branch,
    AVG(amount) AS average_transaction_amount
FROM transactions
GROUP BY branch
ORDER BY average_transaction_amount DESC
LIMIT 1;
